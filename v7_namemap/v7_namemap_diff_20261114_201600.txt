+def _resolve_tag_name(tag, tag_to_brep_name):
+    """tag -> BREP 真名。单个 tag 直接查表；组合 tag（"A#S7+A#S8"，多 solid
+    合并成一个实体）逐个解析：全配上且去实例后缀后同名则取公共件名
+    （如 BMS.1+BMS.2 -> BMS），全配上但不同名则用 " + " 连接，有一部分
+    没配上则返回 None（不猜，保留原 tag）。空 tag（纯分支点）返回 None。"""
+    if not tag:
+        return None
+    # 先整 tag 查（单个 solid 的 tag 本身可能含 "+"）
+    if tag in tag_to_brep_name:
+        return tag_to_brep_name[tag]
+    if "+" not in tag:
+        return None
+    # 组合 tag："base#S7+S8"——后半段是裸 S 编号，需补 base 前缀
+    parts = tag.split("+")
+    base = parts[0].rsplit("#", 1)[0] if "#" in parts[0] else None
+    full = []
+    for i, p in enumerate(parts):
+        if i == 0 or "#" in p:
+            full.append(p)
+        elif base is not None:
+            full.append(base + "#" + p)
+        else:
+            return None
+    names = [tag_to_brep_name.get(f) for f in full]
+    if any(n is None for n in names):
+        return None
+    stripped = [_strip_instance_suffix(n) for n in names]
+    if len(set(stripped)) == 1:
+        return stripped[0]
+    return " + ".join(names)
